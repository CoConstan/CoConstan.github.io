def gazparfait(V, P, R, T):
    n = (P*V)/(R*T)
    return(n)

P = 101325 # 1 atm
T = 293.15 # il fait 20 degrés
R = 8.314
X = np.array([15, 89, 56, 78, 152, 66, 48, 77, 2, 96])

def statdesc(x):
    return {
        "moyenne": np.mean(x),
        "mediane": np.median(x),
        "ecart_type": np.std(x),
        "minimum": np.min(x),
        "maximum": np.max(x)
    }
